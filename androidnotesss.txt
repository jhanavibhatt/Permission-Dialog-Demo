class DownloadFile(
    private val appContext: Context,
    private val onSuccessful: ((String) -> Unit)?,
    private val onFailed: ((String) -> Unit)?
) {
    // ⭐ 1. Define a cancellable Job for the scope ⭐
    private val job = SupervisorJob()

    // Use the Job to define the scope
    private val downloadScope = CoroutineScope(Dispatchers.IO + job)
    private var startTime = System.currentTimeMillis()

    // ⭐ 2. Track the active HttpURLConnection for explicit disconnection ⭐
    private var activeConnection: HttpURLConnection? = null

    private fun tempPathFor(destinationPath: String): String = "$destinationPath.download"

    private fun commitTempFile(tmpFile: File, destinationFile: File): Boolean {
        if (!tmpFile.exists()) return false
        destinationFile.parentFile?.mkdirs()

        // Best-effort remove old destination so rename can succeed.
        if (destinationFile.exists() && !destinationFile.delete()) {
            Timber.tag("DownloadDocService").w("Unable to delete existing destination: ${destinationFile.absolutePath}")
        }

        // Prefer rename (same filesystem).
        if (tmpFile.renameTo(destinationFile)) return destinationFile.exists()

        // Fallback to copy+replace.
        return try {
            tmpFile.copyTo(destinationFile, overwrite = true)
            tmpFile.delete()
            destinationFile.exists()
        } catch (e: Exception) {
            Timber.tag("DownloadDocService").e(e, "Failed to commit temp file to destination")
            false
        }
    }


    fun startDownload(saveFilePath: String, downloadUrl: String, dockey: String = "", docName: String = "") {
        downloadScope.launch {
            try {
                val isSuccess = downloadFile(downloadUrl, saveFilePath)
                withContext(Dispatchers.Main) {
                    if (job.isActive) { // Check if not canceled
                        if (isSuccess) {
                            val duration = System.currentTimeMillis() - startTime
                            Timber.tag("DownloadDocService_3").d("Download Completed in ${duration / 1000} seconds")
                            onSuccessful?.invoke(saveFilePath)
                        } else {
                            if (dockey.isNotEmpty()) {
                                Timber.tag("DownloadDocService_3").e("ReTry")
                                createApi(dockey, Constant.DOC_PATH_ORIGINAL, docName)
                            } else {
                                Timber.tag("DownloadDocService_3").e("Error in Download")
                                onFailed?.invoke("Download failed")
                            }
                        }
                    }
                }
            } catch (e: Exception) {
                e.printStackTrace()

                if (e is CancellationException) throw e // Rethrow for clean shutdown

                if (job.isActive) { // Check if not canceled
                    onFailed?.invoke(e.localizedMessage ?: "Something went wrong during download")
                }
            } finally {
                activeConnection?.disconnect()
                activeConnection = null
            }
        }
    }

    private suspend fun ResponseBody.saveFile(destinationPath: String): Boolean =
        withContext(Dispatchers.IO) {
            val destinationFile = File(destinationPath)
            val tmpFile = File(tempPathFor(destinationPath))
            destinationFile.parentFile?.mkdirs()
            tmpFile.parentFile?.mkdirs()

            var success = false
            try {
                if (tmpFile.exists()) tmpFile.delete()

                tmpFile.outputStream().use { outputStream ->
                    byteStream().use { inputStream ->
                        val buffer = ByteArray(32 * 1024) // 32 KB buffer
                        var bytesRead = 0
                        val startTime = System.currentTimeMillis()

                        while (isActive && inputStream.read(buffer).also { bytesRead = it } != -1) {
                            outputStream.write(buffer, 0, bytesRead)
                        }
                        val duration = System.currentTimeMillis() - startTime
                        if (bytesRead == -1) {
                            Timber.tag("DownloadDocService_3").d("Upload Completed in ${duration / 1000} seconds")
                            outputStream.flush()
                            success = true
                        } else {
                            // This case handles a loop break due to premature cancellation/error
                            Timber.tag("DownloadDocService_3").e("Download aborted before completion.")
                        }
                    }
                }
                if (success) {
                    commitTempFile(tmpFile, destinationFile)
                } else {
                    tmpFile.delete()
                    false
                }

            } catch (e: Exception) {
                tmpFile.delete()
                when (e) {
                    is SocketException if e.message?.contains("connection abort", ignoreCase = true) == true -> {
                        // This is the OS/client-caused abort. Log it but do not crash the app.
                        Timber.tag("DownloadDocService_3").e(e, "SocketException (Connection Abort) during download.")

                        // We return false, allowing higher-level code to retry or handle the failure
                        false
                    }

                    is CancellationException -> {
                        // Always re-throw CancellationException to respect coroutine structure
                        Timber.tag("DownloadDocService_3").w("Download coroutine was cancelled.")
                        throw e
                    }

                    else -> {
                        // Handle all other exceptions (File I/O, general errors)
                        e.printStackTrace()

                        false
                    }
                }
            }
        }

    private suspend fun ResponseBody.saveFileParallel(
        destinationPath: String,
        downloadUrl: String
    ): Boolean = withContext(Dispatchers.IO) {
        val destinationFile = File(destinationPath)
        val tmpFile = File(tempPathFor(destinationPath))
        try {
            destinationFile.parentFile?.mkdirs()
            tmpFile.parentFile?.mkdirs()
            val contentLength = contentLength().takeIf { it > 0 } ?: return@withContext false

            if (tmpFile.exists()) tmpFile.delete()
            RandomAccessFile(tmpFile, "rw").use { it.setLength(contentLength) }

            val semaphore = Semaphore(4)
            val chunkSize = 10 * 1024 * 1024
            val chunkCount = ceil(contentLength.toDouble() / chunkSize).toInt()

            val allChunksOk = coroutineScope {
                (0 until chunkCount).map { chunkIndex ->
                    async {
                        semaphore.withPermit {
                            try {
                                val start = chunkIndex * chunkSize.toLong()
                                val end = min(start + chunkSize, contentLength) - 1

                                val downloadService: RetrofitService = RetrofitApi.getRetrofit()
                                val rangeHeader = "bytes=$start-$end"
                                val response = downloadService.downloadFileWithRange(downloadUrl, rangeHeader)

                                if (response.isSuccessful) {
                                    response.body()?.byteStream()?.use { input ->
                                        RandomAccessFile(tmpFile, "rw").use { output ->
                                            output.seek(start)
                                            val buffer = ByteArray(128 * 1024)
                                            var bytesRead: Int
                                            while (input.read(buffer).also { bytesRead = it } != -1) {
                                                output.write(buffer, 0, bytesRead)
                                            }
                                        }
                                    }
                                    true
                                } else {
                                    Timber.e("Chunk $chunkIndex failed with code ${response.code()}")
                                    false
                                }
                            } catch (e: Exception) {
                                e.printStackTrace()

                                if (e is CancellationException) throw e
                                Timber.e(e, "Chunk $chunkIndex failed")
                                false
                            }
                        }
                    }
                }.awaitAll().all { it }
            }

            if (!allChunksOk) {
                tmpFile.delete()
                return@withContext false
            }

            commitTempFile(tmpFile, destinationFile)
        } catch (e: Exception) {
            e.printStackTrace()

            if (e is CancellationException) throw e
            Timber.e(e, "Parallel download failed")
            tmpFile.delete()
            false
        }
    }

    fun createApi(docKey: String?, docPath: String, docName: String) {
        docKey ?: return
        KotlinUtils.callPresignUrlApi(
            appContext,
            "get",
            docKey,
            docPath,
            { presignedUrlResponse ->

                presignedUrlResponse.annotationUrl.takeIf { it.isNotEmpty() }?.let { annotationUrl ->
                    startDownload(
                        Constant.DownloadPDFPath + docName + Constant.EXTENSION_PDF,
                        annotationUrl,
                        docKey,
                        docName
                    )
                    val duration = System.currentTimeMillis() - startTime
                    Timber.tag("DownloadDocService_3").d("Download Started in ${duration / 1000} seconds")
//                    downloadFileFromURL(docKey,annotationUrl)
                }

                presignedUrlResponse.originalUrl.takeIf { it.isNotEmpty() }?.let { originalUrl ->
                    startDownload(
                        Constant.DownloadPDFPath + docName + Constant.EXTENSION_PDF,
                        originalUrl
                    )
                }
            },
            {
                onFailed?.invoke("No Internet Found!")
            }
        )
    }

    fun downloadFileFromURL(docKey: String, path: String?) {
        downloadScope.launch {
            try {
                val url = URL(path)
                Authenticator.setDefault(object : Authenticator() {
                    override fun getPasswordAuthentication(): PasswordAuthentication {
                        return PasswordAuthentication(
                            ReleaseConstant.fillBetaUsername,
                            ReleaseConstant.fillBetaPassword.toCharArray()
                        )
                    }
                })
                val ucon = url.openConnection() as HttpURLConnection
                activeConnection = ucon // Store connection for cancellation
                ucon.readTimeout = 5000
                ucon.connectTimeout = 10000
                val `is` = ucon.inputStream
                val inStream = BufferedInputStream(`is`, 1024 * 5)
                val destinationFile = File(Constant.DownloadPDFPath + docKey + Constant.EXTENSION_PDF)
                val tmpFile = File(tempPathFor(destinationFile.absolutePath))
                tmpFile.parentFile?.mkdirs()
                if (tmpFile.exists()) tmpFile.delete()

                val outStream = FileOutputStream(tmpFile)
                val buff = ByteArray(5 * 1024)
                var len: Int
                while (inStream.read(buff).also { len = it } != -1) {
                    outStream.write(buff, 0, len)
                }
                outStream.flush()
                outStream.close()
                inStream.close()
                val duration = System.currentTimeMillis() - startTime
                Timber.tag("DownloadDocService").d("Download Completed in ${duration / 1000} seconds")

                if (job.isActive) {
                    if (commitTempFile(tmpFile, destinationFile)) {
                        onSuccessful?.invoke(destinationFile.absolutePath)
                    } else {
                        onFailed?.invoke("Download failed")
                    }
                }
            } catch (e: Exception) {
                e.printStackTrace()

                if (e is CancellationException) throw e
                if (job.isActive) {
                    onFailed?.invoke(e.localizedMessage ?: "Download failed via URL connection")
                }
            } finally {
                activeConnection?.disconnect()
                activeConnection = null
            }
        }
    }

    private suspend fun downloadFile(downloadUrl: String, destinationPath: String): Boolean =
        withContext(Dispatchers.IO) {
            val downloadService: RetrofitService = RetrofitApi.getRetrofit()
            try {
                val response = downloadService.downloadFileWithDynamicUrlSync(downloadUrl)
                if (response.isSuccessful) {
                    val contentLength = response.body()?.contentLength() ?: 0
                    val duration = System.currentTimeMillis() - startTime
                    Timber.tag("DownloadDocService_3").d("Download Completed in ${duration / 1000} seconds) with size ${contentLength / 1024} KB")

                    // Check file size to decide on download method
                    if (contentLength > 1 * 1024 * 1024) { // 1 MB
                        // Use parallel download for large files
                        response.body()?.saveFileParallel(destinationPath, downloadUrl) ?: false
//                        response.body()?.saveFile(destinationPath) ?: false
                    } else {
                        // Use a simple, sequential download for small files
                        response.body()?.saveFile(destinationPath) ?: false
                    }
                } else {
                    false
                }
            } catch (e: Exception) {
                e.printStackTrace()

                if (e is CancellationException) throw e
                false
            }
        }

    /**
     * The explicit cancellation function called by the CoroutineWorker.
     */
    fun cancel() {
        job.cancel()
        activeConnection?.disconnect()
    }

class DownloadDocService(private val context: Context, workerParams: WorkerParameters) :
    CoroutineWorker(context, workerParams) {

    private val documentDao = DocumentRepository(applicationContext)

    /** Rows to persist after PDF/thumbnail work; flushed in one transaction when downloads finish. */
    private val pendingDocumentUpdates =
        Collections.synchronizedList(mutableListOf<Document>())

    private lateinit var preSignedUrls: List<PreSignedUrl>
    private lateinit var documents: List<Document>
    private lateinit var docKey: String

    // Track the currently active download for cancellation
    private var currentDownload: DownloadFile? = null
    private var startTime = System.currentTimeMillis()
    private val memoryClass = (applicationContext.getSystemService(ACTIVITY_SERVICE) as ActivityManager).memoryClass
    val permits = when {
        memoryClass >= 8000 -> 20     // High-end
        memoryClass >= 4000 -> 15     // Mid-range
        else -> 15                    // Low-end
//        else -> 25                    // Low-end
    }
    private val downloadSemaphore = Semaphore(permits = permits)

    override suspend fun doWork(): Result = withContext(Dispatchers.IO) {
        try {
            docKey = inputData.getString("docKey") ?: ""
            Timber.tag("DownloadDocService").d("Worker was in process.== $isActive  == docKey == $docKey == Is Stopped == $isStopped ")
            documents = if (docKey.isEmpty()) {
                documentDao.getAllDocuments()
            } else {
                listOfNotNull(documentDao.getDocument(docKey))
            }
            val jsonArray = buildJsonArray() // New helper function to keep doWork clean
            if (!jsonArray.isEmpty) {
                preSignedUrls = withContext(Dispatchers.IO) {
                    callPresignUrlApiSuspended(applicationContext, "get", jsonArray)
                }
                // 2. Start the main download process, which must also be suspending
                downloadAllDocumentsConcurrently()
            }

            // Check if cancellation was the reason for exiting the loop
            if (isStopped) {
                Result.failure() // or success, based on your logic
            } else {
                Result.success()
            }
        } catch (e: Exception) {
            Timber.tag("DownloadDocService").d("Worker was failed.${e.message}")
            e.printStackTrace()

            if (e is CancellationException) throw e // Rethrow for clean shutdown
            Result.retry()
        }
    }

    companion object {
        const val WORKER_TAG = "DownloadDocWorker"
        const val WORKER_TAG_2 = "DownloadDocWorker_2"
    }

    private suspend fun buildJsonArray(): JsonArray = withContext(Dispatchers.IO) {
        val jsonArray = JsonArray()
        documents.forEach { document ->
            try {
                val docFolder = File(document.originPath)
                if (docKey.isNotEmpty() || (!document.docKey.isNullOrEmpty() && !docFolder.exists())) {
                    val jsonObject = JsonObject().apply {
                        addProperty("docKey", document.docKey)
                        val path = Constant.DOC_PATH_ANNOTATION
                        add("docPath", JsonArray().apply { add(path) })
                    }
                    jsonArray.add(jsonObject)
                }
            } catch (e: Exception) {
                e.printStackTrace()
                Timber.tag("DownloadDocService").d("document list .$documents")
            }
        }
        return@withContext jsonArray
    }

    // A wrapper function, possibly in KotlinUtils or a repository
    suspend fun callPresignUrlApiSuspended(
        context: Context,
        method: String,
        jsonArray: JsonArray
    ): List<PreSignedUrl> = suspendCancellableCoroutine { continuation ->

        KotlinUtils.callPresignUrlApi(
            context, method, jsonArray,
            { preSignedUrls ->
                continuation.resume(preSignedUrls)
            },
            { err ->
                continuation.resumeWithException(Exception("Presign URL API failed: $err"))
            }
        )
    }

    private suspend fun downloadSingleDocument(index: Int, preSigned: PreSignedUrl) =
        suspendCancellableCoroutine { continuation ->
            val doc = documentDao.getDocument(preSigned.docKey) ?: return@suspendCancellableCoroutine

            val saveFilePath = doc.originPath.ifEmpty { Constant.DownloadPDFPath + preSigned.docKey + Constant.EXTENSION_PDF }

            if (File(doc.originPath).exists() && docKey.isEmpty()) {
                Timber.tag("DownloadDocService").e("File already exists: $saveFilePath")
                continuation.resume(Unit)
                return@suspendCancellableCoroutine
            }

            // 1. Create the DownloadFile instance with callbacks that resume/throw
            val downloadFile = DownloadFile(applicationContext, {
                // Success Callback
                CoroutineScope(continuation.context).launch(Dispatchers.IO) {
                    try {
                        // Blocking DAO call is now safely on Dispatchers.IO
                        Timber.tag("DownloadDocService")
                            .e("Success == ${index + 1}/${preSignedUrls.size}  ==   $saveFilePath")
                        readPdfFile(doc, saveFilePath)
                        // Only resume the continuation *after* the blocking call completes
                        continuation.resume(Unit)
                    } catch (e: Exception) {
                        // If the database operation fails, signal failure to the outer coroutine
                        continuation.resumeWithException(Exception("Database update failed: ${e.message}"))
                    }
                }
            }, { err ->
                Timber.tag("DownloadDocService").e("Error $err")
                // Throw an exception to the coroutine flow, which will be caught in downloadAllDocuments.
                continuation.resumeWithException(Exception("Download failed: $err"))
            })

            // CRITICAL: Store the download instance and link cancellation
            currentDownload = downloadFile

            // 2. Link Coroutine Cancellation to DownloadFile Cancellation
            // If the WorkManager job is canceled, this block runs immediately.
            continuation.invokeOnCancellation {
                Timber.tag("DownloadDocService").d("Coroutine canceled, stopping DownloadFile.")
                downloadFile.cancel()
            }

            // 3. Start the download
            when {
//                preSigned.originalUrl.isNotEmpty() -> {
//                    downloadFile.startDownload(saveFilePath, preSigned.originalUrl)
//                }
                preSigned.annotationUrl.isNotEmpty() -> {
                    downloadFile.startDownload(saveFilePath, preSigned.annotationUrl, preSigned.docKey)
                }

                else -> {
                    // No URL, immediately finish this part of the work
                    continuation.resume(Unit)
                }
            }
        }

    private suspend fun downloadAllDocumentsConcurrently() {
        try {
            // Uses coroutineScope to ensure all concurrent downloads complete
            // before this function returns.
            coroutineScope {
                val deferredDownloads = preSignedUrls.mapIndexed { index, preSigned ->
                    // Launch each download concurrently on the IO dispatcher for network operations.
                    async(Dispatchers.IO) {
                        downloadSemaphore.withPermit {
                            try {
                                downloadSingleDocument(index, preSigned)
                                true // Success indicator
                            } catch (e: Exception) {
                                Timber.tag("DownloadDocService").e("Download failed for document $index: ${e.message}")
                                false // Failure indicator
                            }
                        }

                    }
                }

                // Wait for all concurrent downloads to finish.
                val results = deferredDownloads.awaitAll()

                // Log summary results
                val successfulCount = results.count { it }
                val failedCount = results.count { !it }
                Timber.tag("DownloadDocService").i("Concurrent downloads finished: $successfulCount success, $failedCount failed.")
                val duration = (System.currentTimeMillis() - startTime) / 1000
                Timber.tag("DownloadDocService_2").e("Download Completed in ==  ${duration / 60} MIN ${duration % 60} SEC")
            }
        } finally {
            flushPendingDocumentUpdates()
        }
    }

    /** Single Room transaction so LiveData emits once for the whole batch. */
    private suspend fun flushPendingDocumentUpdates() = withContext(Dispatchers.IO) {
        val batch = synchronized(pendingDocumentUpdates) {
            val copy = pendingDocumentUpdates.toList()
            pendingDocumentUpdates.clear()
            copy
        }
        if (batch.isEmpty()) return@withContext
        val distinct = batch.distinctBy { it.recId ?: 0L }
        distinct.forEach { documentDao.updateDocument(it) }
        Timber.tag("DownloadDocService").d("Batch-updated ${distinct.size} document(s) in DB.")
    }

    private suspend fun readPdfFile(document: Document, filePath: String) {
        withContext(Dispatchers.Default) {
            var descriptor: ParcelFileDescriptor? = null
            var pdfRenderer: PdfRenderer? = null
            try {
                descriptor = ParcelFileDescriptor.open(
                    File(filePath),
                    ParcelFileDescriptor.MODE_READ_ONLY
                )
                pdfRenderer = PdfRenderer(descriptor)
//              val pageCount = pdfRenderer.pageCount
//              renderPagesAsync(0, pageCount, filePath, document.recId?:0, pageCount, document, filePath)
                renderPagesAsync(0, 1, filePath, document.recId ?: 0, 1, document, filePath)
            } catch (e: Exception) {
//              deleteFile(document.recId?:0, filePath)
                e.printStackTrace()
            } finally {
                descriptor?.close()
                pdfRenderer?.close()
            }
        }
    }

    private suspend fun renderPagesAsync(
        startPage: Int,
        endPageExclusive: Int,
        src: String,
        recId: Long,
        totalPages: Int,
        document: Document,
        filePath: String
    ) = coroutineScope {
        // Limit parallelism: 2–4 tasks at a time depending on memory
        val permits = when {
            memoryClass >= 8000 -> 6     // High-end
            memoryClass >= 4000 -> 4     // Mid-range
            else -> 2                    // Low-end
        }
        var res = 0
        val semaphore = Semaphore(permits = permits)

        (startPage until endPageExclusive).map { page ->
            async(Dispatchers.IO) {
                semaphore.withPermit {
                    try {
                        res = PdfToImage(
                            src = src,
                            context = context,
                            null,
                            page,
                            recId = recId,
                            totalPages,
                            endPageExclusive - 1
                        ).convertPdfToImage()
                    } catch (e: Exception) {
                        Timber.e(e, "Error rendering page $page")
                    }
                }
            }
        }.awaitAll()
        if (res == 1) {
            document.apply {
                this.docUpdateNow = 0
                this.isLoaded = 0
                this.pageCount = totalPages
                this.originPath = filePath
            }
            pendingDocumentUpdates.add(document)
            Utils.deleteExportedFile(recId)
            // Persist in [flushPendingDocumentUpdates]; Room LiveData refreshes home lists in one shot.
        }
    }

    private suspend fun deleteFile(recId: Long, filePath: String) {
        withContext(Dispatchers.IO) {
//            CommonAPI.deleteFile(context, fileId.orEmpty())
            File(filePath).deleteRecursively()
            documentDao.deleteDocument(recId)
            documentDao.deleteAllAnnotations(recId)
            Utils.deleteLocalFileOfDocument(recId)
        }
    }
